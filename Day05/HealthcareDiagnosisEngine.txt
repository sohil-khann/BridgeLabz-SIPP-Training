// Healthcare Diagnosis Engine
// Concepts: OOPS, Strategy Pattern, Encapsulation, Interfaces

// 1. Define the DiagnosisStrategy Interface
INTERFACE DiagnosisStrategy:
    METHOD diagnose(List<String> symptoms) RETURNS List<String>
        // Abstract method to be implemented by concrete strategies
END INTERFACE

// 2. Implement Concrete Diagnosis Strategies

// RuleBasedDiagnosis: Matches symptoms against predefined rules
CLASS RuleBasedDiagnosis IMPLEMENTS DiagnosisStrategy:
    METHOD diagnose(List<String> symptoms) RETURNS List<String>:
        SET possibleDiseases = new List<String>()

        IF "fever" IN symptoms AND "cough" IN symptoms AND "fatigue" IN symptoms:
            ADD "Flu" to possibleDiseases
            ADD "COVID-19" to possibleDiseases
        END IF
        IF "headache" IN symptoms AND "nausea" IN symptoms:
            ADD "Migraine" to possibleDiseases
        END IF
        // Add more rules as needed

        RETURN possibleDiseases
END CLASS

// AIBasedDiagnosis: Simulates an AI model's diagnosis
CLASS AIBasedDiagnosis IMPLEMENTS DiagnosisStrategy:
    METHOD diagnose(List<String> symptoms) RETURNS List<String>:
        SET possibleDiseases = new List<String>()

        IF "headache" IN symptoms AND "dizziness" IN symptoms AND "nausea" IN symptoms:
            ADD "Migraine" to possibleDiseases
            ADD "Vertigo" to possibleDiseases
        END IF
        IF "chest pain" IN symptoms AND "shortness of breath" IN symptoms:
            ADD "Cardiac Issue" to possibleDiseases
        END IF
        // Simulate complex AI logic

        RETURN possibleDiseases
END CLASS

// ProbabilityBasedDiagnosis: Diagnoses based on symptom probabilities
CLASS ProbabilityBasedDiagnosis IMPLEMENTS DiagnosisStrategy:
    METHOD diagnose(List<String> symptoms) RETURNS List<String>:
        SET possibleDiseases = new List<String>()

        IF "joint pain" IN symptoms AND "swelling" IN symptoms:
            ADD "Arthritis" to possibleDiseases
        END IF
        IF "skin rash" IN symptoms AND "itching" IN symptoms:
            ADD "Dermatitis" to possibleDiseases
        END IF
        // Simulate probability calculations

        RETURN possibleDiseases
END CLASS

// 3. DiagnosisContext: Manages and switches strategies
CLASS DiagnosisContext:
    PRIVATE DiagnosisStrategy currentStrategy

    CONSTRUCTOR(DiagnosisStrategy strategy):
        SET this.currentStrategy = strategy

    METHOD setStrategy(DiagnosisStrategy strategy):
        SET this.currentStrategy = strategy

    METHOD performDiagnosis(List<String> symptoms) RETURNS List<String>:
        PRINT "Performing diagnosis using: " + TYPE_OF(currentStrategy)
        RETURN currentStrategy.diagnose(symptoms)
END CLASS

// 4. Main Program / Scenario Demonstration
PROGRAM Main:
    // Scenario 1: General Medicine patient with RuleBasedDiagnosis
    PRINT "\n--- Scenario 1: General Medicine ---"
    SET ruleBasedStrategy = new RuleBasedDiagnosis()
    SET generalMedicineContext = new DiagnosisContext(ruleBasedStrategy)
    SET symptoms1 = new List<String>("fever", "cough", "fatigue")
    SET diagnosis1 = generalMedicineContext.performDiagnosis(symptoms1)
    PRINT "Symptoms: " + symptoms1
    PRINT "Diagnosis: " + diagnosis1

    // Scenario 2: Neurology patient with AIBasedDiagnosis
    PRINT "\n--- Scenario 2: Neurology ---"
    SET aiBasedStrategy = new AIBasedDiagnosis()
    SET neurologyContext = new DiagnosisContext(aiBasedStrategy)
    SET symptoms2 = new List<String>("headache", "dizziness", "nausea")
    SET diagnosis2 = neurologyContext.performDiagnosis(symptoms2)
    PRINT "Symptoms: " + symptoms2
    PRINT "Diagnosis: " + diagnosis2

    // Demonstrate adding a new algorithm without modifying existing code (plug-and-play)
    // Assume GeneticPatternDiagnosis is a new class implementing DiagnosisStrategy
    CLASS GeneticPatternDiagnosis IMPLEMENTS DiagnosisStrategy:
        METHOD diagnose(List<String> symptoms) RETURNS List<String>:
            SET possibleDiseases = new List<String>()
            IF "genetic marker A" IN symptoms AND "family history B" IN symptoms:
                ADD "Hereditary Condition X" to possibleDiseases
            END IF
            RETURN possibleDiseases
    END CLASS

    PRINT "\n--- Scenario 3: Adding a new algorithm (GeneticPatternDiagnosis) ---"
    SET geneticStrategy = new GeneticPatternDiagnosis()
    // The context can seamlessly switch to the new strategy
    generalMedicineContext.setStrategy(geneticStrategy)
    SET symptoms3 = new List<String>("genetic marker A", "family history B")
    SET diagnosis3 = generalMedicineContext.performDiagnosis(symptoms3)
    PRINT "Symptoms: " + symptoms3
    PRINT "Diagnosis: " + diagnosis3

    // Another example with ProbabilityBasedDiagnosis
    PRINT "\n--- Scenario 4: Using ProbabilityBasedDiagnosis ---"
    SET probabilityStrategy = new ProbabilityBasedDiagnosis()
    SET newContext = new DiagnosisContext(probabilityStrategy)
    SET symptoms4 = new List<String>("joint pain", "swelling")
    SET diagnosis4 = newContext.performDiagnosis(symptoms4)
    PRINT "Symptoms: " + symptoms4
    PRINT "Diagnosis: " + diagnosis4
END PROGRAM
